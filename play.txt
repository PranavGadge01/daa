PLAYFAIR CIPHER
File: PLAY.txt

Playfair encrypts two letters at a time. I and J share one cell.
The program leaves the X/Q filler characters in the decrypted text.

def make_matrix(key):
    alphabet = "ABCDEFGHIKLMNOPQRSTUVWXYZ"  # J is omitted
    letters = ""
    for ch in (key.upper().replace("J", "I") + alphabet):
        if ch in alphabet and ch not in letters:
            letters += ch
    return [list(letters[i:i + 5]) for i in range(0, 25, 5)]


def prepare(text):
    text = "".join(ch for ch in text.upper().replace("J", "I") if ch.isalpha())
    pairs = ""
    i = 0
    while i < len(text):
        a = text[i]
        if i + 1 == len(text):
            b = "X" if a != "X" else "Q"
            i += 1
        elif text[i + 1] == a:
            b = "X" if a != "X" else "Q"
            i += 1
        else:
            b = text[i + 1]
            i += 2
        pairs += a + b
    return pairs


def find_position(matrix, letter):
    for row in range(5):
        for col in range(5):
            if matrix[row][col] == letter:
                return row, col


key = input("Enter keyword: ")
matrix = make_matrix(key)
print("5 x 5 key matrix:")
for row in matrix:
    print(" ".join(row))

action = input("Encrypt or decrypt? (E/D): ").upper()
text = input("Enter text: ")

if action == "E":
    text = prepare(text)
else:
    text = "".join(ch for ch in text.upper().replace("J", "I") if ch.isalpha())
    if len(text) % 2 != 0:
        raise ValueError("Ciphertext must contain an even number of letters.")

result = ""
step = 1 if action == "E" else -1
for i in range(0, len(text), 2):
    a, b = text[i], text[i + 1]
    r1, c1 = find_position(matrix, a)
    r2, c2 = find_position(matrix, b)

    if r1 == r2:
        result += matrix[r1][(c1 + step) % 5]
        result += matrix[r2][(c2 + step) % 5]
    elif c1 == c2:
        result += matrix[(r1 + step) % 5][c1]
        result += matrix[(r2 + step) % 5][c2]
    else:
        result += matrix[r1][c2]
        result += matrix[r2][c1]

print("Result:", result)

import java.util.Scanner;

class Main {
    static char[][] makeMatrix(String key) {
        String alphabet = "ABCDEFGHIKLMNOPQRSTUVWXYZ";
        String letters = "";
        key = key.toUpperCase().replace('J', 'I');
        String all = key + alphabet;

        for (int i = 0; i < all.length(); i++) {
            char ch = all.charAt(i);
            if (alphabet.indexOf(ch) >= 0 && letters.indexOf(ch) < 0) {
                letters += ch;
            }
        }

        char[][] matrix = new char[5][5];
        for (int i = 0; i < 25; i++) {
            matrix[i / 5][i % 5] = letters.charAt(i);
        }
        return matrix;
    }

    static int[] findPosition(char[][] matrix, char ch) {
        for (int r = 0; r < 5; r++) {
            for (int c = 0; c < 5; c++) {
                if (matrix[r][c] == ch) return new int[]{r, c};
            }
        }
        return new int[]{-1, -1};
    }

    static String prepare(String text) {
        text = text.toUpperCase().replace('J', 'I').replaceAll("[^A-Z]", "");
        String result = "";
        int i = 0;
        while (i < text.length()) {
            char a = text.charAt(i);
            char b;
            if (i + 1 == text.length()) {
                b = (a == 'X') ? 'Q' : 'X';
                i++;
            } else if (text.charAt(i + 1) == a) {
                b = (a == 'X') ? 'Q' : 'X';
                i++;
            } else {
                b = text.charAt(i + 1);
                i += 2;
            }
            result += a;
            result += b;
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter keyword: ");
        String key = sc.nextLine();
        char[][] matrix = makeMatrix(key);

        System.out.println("5 x 5 key matrix:");
        for (int r = 0; r < 5; r++) {
            for (int c = 0; c < 5; c++) System.out.print(matrix[r][c] + " ");
            System.out.println();
        }

        System.out.print("Encrypt or decrypt? (E/D): ");
        String action = sc.nextLine().toUpperCase();
        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase().replace('J', 'I').replaceAll("[^A-Z]", "");

        if (action.equals("E")) text = prepare(text);
        else if (text.length() % 2 != 0) {
            System.out.println("Ciphertext must contain an even number of letters.");
            sc.close();
            return;
        }

        int step = action.equals("E") ? 1 : -1;
        String result = "";
        for (int i = 0; i < text.length(); i += 2) {
            char a = text.charAt(i), b = text.charAt(i + 1);
            int[] p1 = findPosition(matrix, a);
            int[] p2 = findPosition(matrix, b);
            int r1 = p1[0], c1 = p1[1], r2 = p2[0], c2 = p2[1];

            if (r1 == r2) {
                result += matrix[r1][(c1 + step + 5) % 5];
                result += matrix[r2][(c2 + step + 5) % 5];
            } else if (c1 == c2) {
                result += matrix[(r1 + step + 5) % 5][c1];
                result += matrix[(r2 + step + 5) % 5][c2];
            } else {
                result += matrix[r1][c2];
                result += matrix[r2][c1];
            }
        }
        System.out.println("Result: " + result);
        sc.close();
    }
}
