EULER PRIMALITY TEST (EULER-JACOBI)
File: EULR.txt

For an odd n and a base a coprime to n, compare a^((n-1)/2) mod n with
the Jacobi symbol (a/n). The Jacobi symbol is implemented below.
A passing result means PROBABLY PRIME, not a proof of primality.

PYTHON CODE
-----------
import random
import math


def jacobi(a, n):
    result = 1
    a = a % n
    while a != 0:
        while a % 2 == 0:
            a //= 2
            if n % 8 in (3, 5):
                result = -result
        a, n = n, a
        if a % 4 == 3 and n % 4 == 3:
            result = -result
        a %= n
    return result if n == 1 else 0


n = int(input("Enter number: "))
rounds = 5

if n < 2:
    print("Composite")
elif n in (2, 3):
    print("Prime")
elif n % 2 == 0:
    print("Composite")
else:
    probably_prime = True
    for i in range(rounds):
        a = random.randint(2, n - 2)
        if math.gcd(a, n) != 1:
            probably_prime = False
            break
        j = jacobi(a, n)
        expected = j % n
        actual = pow(a, (n - 1) // 2, n)
        if actual != expected:
            probably_prime = False
            break
    print("Probably Prime" if probably_prime else "Composite")

JAVA CODE
---------
import java.math.BigInteger;
import java.util.Random;
import java.util.Scanner;

class Main {
    static int jacobi(int a, int n) {
        int result = 1;
        a = a % n;
        while (a != 0) {
            while (a % 2 == 0) {
                a /= 2;
                if (n % 8 == 3 || n % 8 == 5) result = -result;
            }
            int temp = a;
            a = n;
            n = temp;
            if (a % 4 == 3 && n % 4 == 3) result = -result;
            a %= n;
        }
        return n == 1 ? result : 0;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Random random = new Random();
        System.out.print("Enter number: ");
        int n = sc.nextInt();
        int rounds = 5;

        if (n < 2) {
            System.out.println("Composite");
        } else if (n == 2 || n == 3) {
            System.out.println("Prime");
        } else if (n % 2 == 0) {
            System.out.println("Composite");
        } else {
            boolean probablyPrime = true;
            BigInteger N = BigInteger.valueOf(n);
            for (int i = 0; i < rounds; i++) {
                int a = random.nextInt(n - 3) + 2;
                if (!BigInteger.valueOf(a).gcd(N).equals(BigInteger.ONE)) {
                    probablyPrime = false;
                    break;
                }
                int j = jacobi(a, n);
                int expected = (j == -1) ? n - 1 : j;
                int actual = BigInteger.valueOf(a)
                        .modPow(BigInteger.valueOf((n - 1L) / 2), N).intValue();
                if (actual != expected) {
                    probablyPrime = false;
                    break;
                }
            }
            System.out.println(probablyPrime ? "Probably Prime" : "Composite");
        }
        sc.close();
    }
}
